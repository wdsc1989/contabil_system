# OU se o script não existir, execute este código Python diretamente:

cd /opt/contabil
source venv/bin/activate
python3 << 'EOF'
from config.database import engine
from sqlalchemy import text

print("Adicionando colunas...")
with engine.begin() as conn:
    # Contracts
    for col in ['seller_name VARCHAR(200)', 'event_location TEXT', 'service_hours FLOAT', 
                'collaborators TEXT', 'invoice_number VARCHAR(50)', 'notes TEXT']:
        try:
            conn.execute(text(f"ALTER TABLE contracts ADD COLUMN {col}"))
            print(f"✅ contracts.{col.split()[0]}")
        except Exception as e:
            if "already exists" in str(e) or "duplicate" in str(e).lower():
                print(f"⚠️  contracts.{col.split()[0]} já existe")
            else:
                print(f"❌ contracts.{col.split()[0]}: {e}")
    
    # Accounts Payable
    for col in ['expense_type VARCHAR(20)', 'expense_category VARCHAR(50)', 'description TEXT']:
        try:
            conn.execute(text(f"ALTER TABLE accounts_payable ADD COLUMN {col}"))
            print(f"✅ accounts_payable.{col.split()[0]}")
        except Exception as e:
            if "already exists" in str(e) or "duplicate" in str(e).lower():
                print(f"⚠️  accounts_payable.{col.split()[0]} já existe")
            else:
                print(f"❌ accounts_payable.{col.split()[0]}: {e}")
    
    # Accounts Receivable
    try:
        conn.execute(text("ALTER TABLE accounts_receivable ADD COLUMN contract_id INTEGER"))
        print("✅ accounts_receivable.contract_id")
    except Exception as e:
        if "already exists" in str(e) or "duplicate" in str(e).lower():
            print("⚠️  accounts_receivable.contract_id já existe")
        else:
            print(f"❌ accounts_receivable.contract_id: {e}")

print("\n✅ Concluído! Reinicie o serviço: sudo systemctl restart contabil")
EOF

sudo systemctl restart contabil
